# Copy-paste into your shell, then run:
#   combine_model_009_2_dataset
#
# What this does:
# - merges the base dataset `Combination/sourccey-shirt-fold-c-008`
# - includes every nested dataset root under `Combination/Model-009-datasets`
# - writes the merged output to `Combination/Model-009-2-dataset`
# - runs info + consistency audit + consistency fix + final info

combine_model_009_2_dataset() {
  set -euo pipefail

  HF_LEROBOT_HOME="/home/sourccey/.cache/huggingface/lerobot"
  DATASET_REPO="Combination/Model-009-2-dataset"
  DATASET_ROOT="${HF_LEROBOT_HOME}/${DATASET_REPO}"
  BASE_DATASET_ROOT="${HF_LEROBOT_HOME}/Combination/sourccey-shirt-fold-c-008"
  MODEL_009_PARENT="${HF_LEROBOT_HOME}/Combination/Model-009-datasets"
  STAMP="$(date +%Y%m%d_%H%M%S)"
  SAFE_DATASET_REPO="${DATASET_REPO//\//__}"
  AUDIT_DIR="outputs/video_audit_${STAMP}_full"

  echo "Merging into: ${DATASET_ROOT}"
  echo "Base dataset: ${BASE_DATASET_ROOT}"
  echo "Additional nested datasets parent: ${MODEL_009_PARENT}"

  uv run python src/lerobot/robots/sourccey/sourccey/sourccey/readme/combine.py \
    --parent "${BASE_DATASET_ROOT}" \
    --parent "${MODEL_009_PARENT}" \
    --out-repo "${DATASET_REPO}"

  uv run lerobot-edit-dataset \
    --repo_id "${DATASET_REPO}" \
    --root "${DATASET_ROOT}" \
    --operation.type info \
    --operation.show_features true

  uv run python src/lerobot/robots/sourccey/sourccey/sourccey/readme/audit_dataset_consistency.py \
    --dataset-root "${DATASET_ROOT}"

  uv run python src/lerobot/robots/sourccey/sourccey/sourccey/readme/fix_dataset_consistency.py \
    --dataset-root "${DATASET_ROOT}"

  uv run python src/lerobot/robots/sourccey/sourccey/sourccey/readme/audit_dataset_consistency.py \
    --dataset-root "${DATASET_ROOT}" \
    --fail-on-issues

  uv run lerobot-audit-video-decode-errors \
    --dataset-repo-id "${DATASET_REPO}" \
    --workers 8 \
    --output-dir "${AUDIT_DIR}"

  uv run lerobot-edit-dataset \
    --repo_id "${DATASET_REPO}" \
    --root "${DATASET_ROOT}" \
    --operation.type info \
    --operation.show_features true

  echo
  echo "Done."
  echo "Dataset repo: ${DATASET_REPO}"
  echo "Dataset root: ${DATASET_ROOT}"
  echo "Video audit dir: ${AUDIT_DIR}"
  echo "Bad files list: ${AUDIT_DIR}/${SAFE_DATASET_REPO}_bad_files.txt"
}
